  
-- ============================================  
-- 1. sp_CreateAccount - Creates new account with PENDING status  
-- ============================================  
CREATE   PROCEDURE sp_CreateAccount  
    @CustomerID INT,  
    @AccountClassID INT,  
    @CurrencyID INT,  
    @BranchID INT,  
    @MakerID INT,  
    @AccountID INT OUTPUT,  
    @AccountNumber NVARCHAR(19) OUTPUT  
AS  
BEGIN  
    SET NOCOUNT ON;  
      
    BEGIN TRY  
        BEGIN TRANSACTION;  
  
        -- 1. Get Flex Branch (3 characters)  
        DECLARE @FlexBranch NVARCHAR(3);  
        SELECT @FlexBranch = FlexBranch FROM LKP_Branch WHERE BranchID = @BranchID;  
          
        -- 2. Get Account Class Code (5 characters)  
        DECLARE @AccountClassCode NVARCHAR(5);  
        SELECT @AccountClassCode = Code FROM LKP_AccountClass WHERE AccountClassID = @AccountClassID;  
          
        -- 3. Get last 5 digits of Customer CIF (CIF format: CIF00004 -> 00004)  
        DECLARE @CIFDigits NVARCHAR(5);  
        SELECT @CIFDigits = RIGHT(CIF, 5) FROM Customer WHERE CustomerID = @CustomerID;  
          
        -- 4. Generate 6 random digits (0-9)  
DECLARE @RandomDigits NVARCHAR(6);  
  
SET @RandomDigits =  
    RIGHT(  
        '000000' + CAST(ABS(CHECKSUM(NEWID())) % 1000000 AS NVARCHAR(6)),  
        6  
    );  
          
        -- 5. Build Account Number: FlexBranch + AccountClassCode + CIFDigits + RandomDigits  
        SET @AccountNumber = @FlexBranch + @AccountClassCode + @CIFDigits + @RandomDigits;  
  
        -- 6. Insert Account with PENDING status (StatusID = 3)  
        INSERT INTO Account (  
            CustomerID,  
            AccountClassID,  
            StatusID,  
            AccountNumber,  
            CurrencyID,  
            BranchID,  
            DateCreated,  
            MakerID,  
            MakerDate,  
            CheckerID,  
            CheckerDate,  
            Balance  
        )  
        VALUES (  
            @CustomerID,  
            @AccountClassID,  
            3,  -- PENDING  
            @AccountNumber,  
            @CurrencyID,  
            @BranchID,  
            GETDATE(),  
            @MakerID,  
            GETDATE(),  
            NULL,  
            NULL,  
            0  -- Initial balance  
        );  
          
        SET @AccountID = SCOPE_IDENTITY();  
  
        COMMIT TRANSACTION;  
        RETURN 0;  
    END TRY  
    BEGIN CATCH  
        ROLLBACK TRANSACTION;  
        DECLARE @ErrorMessage NVARCHAR(4000) = ERROR_MESSAGE();  
        RAISERROR(@ErrorMessage, 16, 1);  
        RETURN -1;  
    END CATCH  
END  
  
CREATE PROCEDURE sp_CreateCustomer    
    @Username NVARCHAR(50),    
    @Password NVARCHAR(255),  
    @FullName NVARCHAR(150),    
    @Address NVARCHAR(250),    
    @PhoneNumber NVARCHAR(20),    
    @NationalID NVARCHAR(20),    
    @Email NVARCHAR(100),    
    @MakerID INT,    
    @OnlineRegister BIT,  
    @CustomerID INT OUTPUT,    
    @UserID INT OUTPUT    
AS    
BEGIN    
    SET NOCOUNT ON;    
        
    BEGIN TRY    
        BEGIN TRANSACTION;    
    
        -- Create User account with IsActive = 0 (pending approval)  
        -- Both online and Maker-created customers start as inactive  
        INSERT INTO [User] (RoleID, Username, Password, IsActive)    
        VALUES (1, @Username, @Password, 0);  -- IsActive = 0 for ALL new customers  
            
        SET @UserID = SCOPE_IDENTITY();    
    
        -- Generate CIF number    
        DECLARE @CIF NVARCHAR(20);    
        SET @CIF = 'CIF' + RIGHT('00000' + CAST(@UserID AS NVARCHAR(5)), 5);    
    
        -- Create Customer with OnlineRegister flag and MakerID  
        INSERT INTO Customer (    
            UserID, CIF, FullName, Address, PhoneNumber,     
            NationalID, Email, MakerID, OnlineRegister, DateCreated    
        )    
        VALUES (    
            @UserID, @CIF, @FullName, @Address, @PhoneNumber,    
            @NationalID, @Email, @MakerID, @OnlineRegister, GETDATE()  
        );    
            
        SET @CustomerID = SCOPE_IDENTITY();    
    
        COMMIT TRANSACTION;    
            
        RETURN 0; -- Success    
    END TRY    
    BEGIN CATCH    
        ROLLBACK TRANSACTION;    
        DECLARE @ErrorMessage NVARCHAR(4000) = ERROR_MESSAGE();    
        RAISERROR(@ErrorMessage, 16, 1);    
        RETURN -1; -- Error    
    END CATCH    
END
  
CREATE PROCEDURE sp_CreateLoan  
    @CustomerID INT,  
    @LoanAmount DECIMAL(18,2),  
    @InterestRate DECIMAL(5,2),  
    @TermMonths INT,  
    @MakerID INT,  
    @DisbursementAccountID INT,  -- NEW: Account to receive loan  
    @LoanID INT OUTPUT  
AS  
BEGIN  
    SET NOCOUNT ON;  
      
    BEGIN TRY  
        BEGIN TRANSACTION;  
  
        -- Validate Customer exists and IsActive = 1  
        DECLARE @IsActive BIT;  
        SELECT @IsActive = u.IsActive  
        FROM Customer c  
        INNER JOIN [User] u ON c.UserID = u.UserID  
        WHERE c.CustomerID = @CustomerID;  
          
        IF @IsActive != 1  
        BEGIN  
            RAISERROR('Customer account is not active.', 16, 1);  
            RETURN -1;  
        END  
  
        -- Validate Account belongs to Customer and is Active (StatusID 1 or 4)  
        DECLARE @AccountExists INT;  
        SELECT @AccountExists = COUNT(*)   
        FROM Account   
        WHERE AccountID = @DisbursementAccountID   
        AND CustomerID = @CustomerID   
        AND StatusID IN (1, 4);  -- Approved or Active  
  
        IF @AccountExists = 0  
        BEGIN  
            RAISERROR('Invalid account or account does not belong to this customer.', 16, 1);  
            RETURN -1;  
        END  
  
        -- Insert Loan with PENDING status (StatusID = 3)  
        INSERT INTO Loan (  
            CustomerID,  
            LoanAmount,  
            InterestRate,  
            TermMonths,  
            StatusID,  
            MakerID,  
            CheckerID,  
            DateRequested,  
            DisbursementAccountID  
        )  
        VALUES (  
            @CustomerID,  
            @LoanAmount,  
            @InterestRate,  
            @TermMonths,  
            3,  -- PENDING  
            @MakerID,  
            NULL,  
            GETDATE(),  
            @DisbursementAccountID  
        );  
          
        SET @LoanID = SCOPE_IDENTITY();  
  
        COMMIT TRANSACTION;  
        RETURN 0;  
    END TRY  
    BEGIN CATCH  
        ROLLBACK TRANSACTION;  
        DECLARE @ErrorMessage NVARCHAR(4000) = ERROR_MESSAGE();  
        RAISERROR(@ErrorMessage, 16, 1);  
        RETURN -1;  
    END CATCH  
END  
  
CREATE PROCEDURE sp_CreateTransaction  
    @AccountIDFrom INT,  
    @AccountIDTo INT,  
    @Amount DECIMAL(18,2),  
    @TransactionID INT OUTPUT  
AS  
BEGIN  
    SET NOCOUNT ON;  
      
    BEGIN TRY  
        BEGIN TRANSACTION;  
  
        -- Validate accounts exist  
        DECLARE @FromExists INT, @ToExists INT;  
        SELECT @FromExists = COUNT(*) FROM Account WHERE AccountID = @AccountIDFrom;  
        SELECT @ToExists = COUNT(*) FROM Account WHERE AccountID = @AccountIDTo;  
  
        IF @FromExists = 0 OR @ToExists = 0  
        BEGIN  
            RAISERROR('Invalid account(s).', 16, 1);  
            RETURN -1;  
        END  
  
        -- Check if accounts are the same  
        IF @AccountIDFrom = @AccountIDTo  
        BEGIN  
            RAISERROR('Cannot transfer to the same account.', 16, 1);  
            RETURN -1;  
        END  
  
        -- Get account currencies and balances  
        DECLARE @FromCurrencyID INT, @ToCurrencyID INT;  
        DECLARE @FromBalance DECIMAL(18,2);  
          
        SELECT   
            @FromCurrencyID = CurrencyID,  
            @FromBalance = Balance  
        FROM Account   
        WHERE AccountID = @AccountIDFrom;  
          
        SELECT @ToCurrencyID = CurrencyID  
        FROM Account   
        WHERE AccountID = @AccountIDTo;  
  
        -- Check sufficient balance  
        IF @FromBalance < @Amount  
        BEGIN  
            RAISERROR('Insufficient balance.', 16, 1);  
            RETURN -1;  
        END  
  
        -- Initialize variables  
        DECLARE @FinalAmount DECIMAL(18,2) = @Amount;  
        DECLARE @ExchangeRate DECIMAL(18,6) = NULL;  
        DECLARE @OriginalAmount DECIMAL(18,2) = @Amount;  
        DECLARE @ConvertedAmount DECIMAL(18,2) = @Amount;  
  
        -- If currencies are different, get exchange rate and convert  
        IF @FromCurrencyID != @ToCurrencyID  
        BEGIN  
            -- Get exchange rate from CurrencyRate table  
            SELECT TOP 1 @ExchangeRate = Rate  
            FROM LKP_CurrencyRate  
            WHERE FromCurrencyID = @FromCurrencyID  
            AND ToCurrencyID = @ToCurrencyID;  
  
            IF @ExchangeRate IS NULL  
            BEGIN  
                RAISERROR('Exchange rate not available for these currencies.', 16, 1);  
                RETURN -1;  
            END  
  
            -- Convert amount  
            SET @ConvertedAmount = @Amount * @ExchangeRate;  
            SET @FinalAmount = @ConvertedAmount;  
        END  
  
        -- Update balances  
        UPDATE Account SET Balance = Balance - @Amount WHERE AccountID = @AccountIDFrom;  
        UPDATE Account SET Balance = Balance + @FinalAmount WHERE AccountID = @AccountIDTo;  
  
        -- Create transaction record with exchange info  
        INSERT INTO [Transaction] (  
            AccountIDFrom,   
            AccountIDTo,   
            Amount,   
            DateCreated,  
            ExchangeRate,  
            OriginalAmount,  
            ConvertedAmount  
        )  
        VALUES (  
            @AccountIDFrom,   
            @AccountIDTo,   
            @FinalAmount,   
            GETDATE(),  
            @ExchangeRate,  
            @OriginalAmount,  
            @ConvertedAmount  
        );  
  
        SET @TransactionID = SCOPE_IDENTITY();  
  
        COMMIT TRANSACTION;  
        RETURN 0;  
    END TRY  
    BEGIN CATCH  
        ROLLBACK TRANSACTION;  
        DECLARE @ErrorMessage NVARCHAR(4000) = ERROR_MESSAGE();  
        RAISERROR(@ErrorMessage, 16, 1);  
        RETURN -1;  
    END CATCH  
END  
  
CREATE PROCEDURE sp_GetAccountByID    
    @AccountID INT    
AS    
BEGIN    
    SET NOCOUNT ON;    
        
    SELECT     
        a.AccountID,    
        a.AccountNumber,    
        a.Balance,    
        a.DateCreated,    
        a.MakerDate,    
        a.CheckerDate,    
        c.CIF,    
        c.FullName AS CustomerName,    
        c.CustomerID,    
        ac.AccountClassID,    
        ac.AccountClassName,    
        s.StatusID,    
        s.StatusName,    
        cur.CurrencyID,    
        cur.CurrencyName,    
        b.BranchID,    
        b.BranchName,    
        u.Username AS MakerName,    
        checker.Username AS CheckerName,    
        a.MakerID,    
        a.CheckerID,    
        a.StatusID    
    FROM Account a    
    INNER JOIN Customer c ON a.CustomerID = c.CustomerID    
    INNER JOIN LKP_AccountClass ac ON a.AccountClassID = ac.AccountClassID    
    INNER JOIN LKP_Status s ON a.StatusID = s.StatusID    
    INNER JOIN LKP_Currency cur ON a.CurrencyID = cur.CurrencyID    
    INNER JOIN LKP_Branch b ON a.BranchID = b.BranchID    
    LEFT JOIN [User] u ON a.MakerID = u.UserID    
    LEFT JOIN [User] checker ON a.CheckerID = checker.UserID    
    WHERE a.AccountID = @AccountID;    
END

-- 2. sp_GetTransactionsByAccount - Get transactions for an account  
-- ============================================  
CREATE   PROCEDURE sp_GetTransactionsByAccount  
    @AccountID INT  
AS  
BEGIN  
    SET NOCOUNT ON;  
      
    SELECT   
        t.TransactionID,  
        t.Amount,  
        t.DateCreated,  
        t.AccountIDFrom,  
        t.AccountIDTo,  
        aFrom.AccountNumber AS FromAccountNumber,  
        aTo.AccountNumber AS ToAccountNumber,  
        aFrom.CustomerID AS FromCustomerID,  
        aTo.CustomerID AS ToCustomerID  
    FROM [Transaction] t  
    INNER JOIN Account aFrom ON t.AccountIDFrom = aFrom.AccountID  
    INNER JOIN Account aTo ON t.AccountIDTo = aTo.AccountID  
    WHERE t.AccountIDFrom = @AccountID OR t.AccountIDTo = @AccountID  
    ORDER BY t.DateCreated DESC;  
END  


-- ============================================  
-- 3. sp_GetTransactionsByCustomer - Get all transactions for a customer  
-- ============================================  
CREATE   PROCEDURE sp_GetTransactionsByCustomer  
    @CustomerID INT  
AS  
BEGIN  
    SET NOCOUNT ON;  
      
    SELECT   
        t.TransactionID,  
        t.Amount,  
        t.DateCreated,  
        t.AccountIDFrom,  
        t.AccountIDTo,  
        aFrom.AccountNumber AS FromAccountNumber,  
        aTo.AccountNumber AS ToAccountNumber,  
        aFrom.CustomerID AS FromCustomerID,  
        aTo.CustomerID AS ToCustomerID,  
        CASE   
            WHEN aFrom.CustomerID = @CustomerID THEN 'Sent'  
            WHEN aTo.CustomerID = @CustomerID THEN 'Received'  
            ELSE 'Other'  
        END AS TransactionType  
    FROM [Transaction] t  
    INNER JOIN Account aFrom ON t.AccountIDFrom = aFrom.AccountID  
    INNER JOIN Account aTo ON t.AccountIDTo = aTo.AccountID  
    WHERE aFrom.CustomerID = @CustomerID OR aTo.CustomerID = @CustomerID  
    ORDER BY t.DateCreated DESC;  
END  

CREATE   PROCEDURE sp_UpdateAccountStatus  
    @AccountID INT,  
    @StatusID INT,  
    @CheckerID INT  
AS  
BEGIN  
    SET NOCOUNT ON;  
      
    BEGIN TRY  
        BEGIN TRANSACTION;  
  
        UPDATE Account   
        SET StatusID = @StatusID,  
            CheckerID = @CheckerID,  
            CheckerDate = GETDATE()  
        WHERE AccountID = @AccountID;  
  
        COMMIT TRANSACTION;  
        RETURN 0;  
    END TRY  
    BEGIN CATCH  
        ROLLBACK TRANSACTION;  
        DECLARE @ErrorMessage NVARCHAR(4000) = ERROR_MESSAGE();  
        RAISERROR(@ErrorMessage, 16, 1);  
        RETURN -1;  
    END CATCH  
END  

  
CREATE PROCEDURE sp_UpdateCustomer    
    @CustomerID INT,    
    @FullName NVARCHAR(150),    
    @Address NVARCHAR(250),    
    @PhoneNumber NVARCHAR(20),    
    @NationalID NVARCHAR(20),    
    @Email NVARCHAR(100),    
    @MakerID INT    
AS    
BEGIN    
    SET NOCOUNT ON;    
        
    BEGIN TRY    
        BEGIN TRANSACTION;    
  
        -- Get current IsActive status from User table  
        DECLARE @UserID INT;  
        DECLARE @IsActive BIT;  
          
        SELECT @UserID = UserID FROM Customer WHERE CustomerID = @CustomerID;  
        SELECT @IsActive = IsActive FROM [User] WHERE UserID = @UserID;  
  
        -- Customer can only be edited if NOT active (pending = IsActive = 0)  
        -- OR if they are On Hold (CheckerID is not null but IsActive = 0)  
        DECLARE @CheckerID INT;  
        SELECT @CheckerID = CheckerID FROM Customer WHERE CustomerID = @CustomerID;  
  
        IF @IsActive = 1  
        BEGIN    
            RAISERROR('Active customers cannot be edited.', 16, 1);    
            RETURN -1;    
        END    
    
        -- Update customer details    
        UPDATE Customer     
        SET FullName = @FullName,    
            Address = @Address,    
            PhoneNumber = @PhoneNumber,    
            NationalID = @NationalID,    
            Email = @Email,    
            MakerID = @MakerID,    
            CheckerID = NULL,    
            CheckerDate = NULL    
        WHERE CustomerID = @CustomerID;    
    
        -- Keep IsActive = 0 (Pending) for re-review  
        -- User remains inactive until Checker approves again  
    
        COMMIT TRANSACTION;    
        RETURN 0;    
    END TRY    
    BEGIN CATCH    
        ROLLBACK TRANSACTION;    
        DECLARE @ErrorMessage NVARCHAR(4000) = ERROR_MESSAGE();    
        RAISERROR(@ErrorMessage, 16, 1);    
        RETURN -1;    
    END CATCH    
END  

  
CREATE PROCEDURE sp_UpdateCustomerStatus    
    @CustomerID INT,    
    @StatusID INT,    
    @CheckerID INT = NULL,    
    @MakerID INT = NULL    
AS    
BEGIN    
    SET NOCOUNT ON;    
        
    BEGIN TRY    
        BEGIN TRANSACTION;    
  
        -- Get the UserID for this customer  
        DECLARE @UserID INT;  
        SELECT @UserID = UserID FROM Customer WHERE CustomerID = @CustomerID;  
  
        -- StatusID 1 = Approved -> IsActive = 1  
        -- All other statuses -> IsActive = 0  
        DECLARE @IsActive BIT;  
        SET @IsActive = CASE   
            WHEN @StatusID = 1 THEN 1  -- Approved  
            ELSE 0  -- Rejected, Pending, On Hold  
        END;  
  
        -- Update User IsActive  
        UPDATE [User]   
        SET IsActive = @IsActive  
        WHERE UserID = @UserID;  
  
        -- Update Customer with Checker info  
        UPDATE Customer   
        SET CheckerID = @CheckerID,  
            CheckerDate = GETDATE()  
        WHERE CustomerID = @CustomerID;    
  
        COMMIT TRANSACTION;    
        RETURN 0;    
    END TRY    
    BEGIN CATCH    
        ROLLBACK TRANSACTION;    
        DECLARE @ErrorMessage NVARCHAR(4000) = ERROR_MESSAGE();    
        RAISERROR(@ErrorMessage, 16, 1);    
        RETURN -1;    
    END CATCH    
END;  

  
CREATE PROCEDURE sp_UpdateLoanStatus  
    @LoanID INT,  
    @StatusID INT,  
    @CheckerID INT  
AS  
BEGIN  
    SET NOCOUNT ON;  
      
    BEGIN TRY  
        BEGIN TRANSACTION;  
  
        -- Get loan details  
        DECLARE @CurrentStatusID INT;  
        DECLARE @CustomerID INT;  
        DECLARE @LoanAmount DECIMAL(18,2);  
        DECLARE @DisbursementAccountID INT;  
  
        SELECT   
            @CurrentStatusID = StatusID,  
            @CustomerID = CustomerID,  
            @LoanAmount = LoanAmount,  
            @DisbursementAccountID = DisbursementAccountID  
        FROM Loan   
        WHERE LoanID = @LoanID;  
  
        -- Only allow updates for Pending loans  
        IF @CurrentStatusID != 3  
        BEGIN  
            RAISERROR('Only pending loans can be reviewed.', 16, 1);  
            RETURN -1;  
        END  
  
        -- Update loan status  
        UPDATE Loan   
        SET StatusID = @StatusID,  
            CheckerID = @CheckerID,  
            CheckerDate = GETDATE()  
        WHERE LoanID = @LoanID;  
  
        -- ==============================================  
        -- IF APPROVED: Disburse loan to selected account  
        -- ==============================================  
        IF @StatusID = 1 AND @DisbursementAccountID IS NOT NULL  
        BEGIN  
            -- Add loan amount to account balance  
            UPDATE Account   
            SET Balance = Balance + @LoanAmount  
            WHERE AccountID = @DisbursementAccountID;  
  
            -- Create transaction record (loan disbursement)  
            INSERT INTO [Transaction] (  
                AccountIDFrom,   
                AccountIDTo,   
                Amount,   
                DateCreated  
            )  
            VALUES (  
                NULL,  -- No sender for loan  
                @DisbursementAccountID,   
                @LoanAmount,   
                GETDATE()  
            );  
        END  
  
        COMMIT TRANSACTION;  
        RETURN 0;  
    END TRY  
    BEGIN CATCH  
        ROLLBACK TRANSACTION;  
        DECLARE @ErrorMessage NVARCHAR(4000) = ERROR_MESSAGE();  
        RAISERROR(@ErrorMessage, 16, 1);  
        RETURN -1;  
    END CATCH  
END
